% ============ RESEARCH & PROJECT EXPERIENCE ============
\section{Research \& Project Experience}

\begin{enumerate}[leftmargin=22pt]
    \item \textbf{PhD Research: Physics-Informed Deep Learning for Urban Weather Nowcasting and Emulation} \hfill 2024--Now\\
    Wuhan University, School of Water Resources and Hydropower Engineering
    \begin{itemize}
        \item \textbf{Causal ML for urban expansion--heatwave nexus.} Used double machine learning to decouple urbanization-induced thermal penalties from climate variability across 1,690 Chinese cities (1990--2022); revealed a dichotomy between internal 3D densification (local intensity/onset) and intercity coalescence (regional frequency/duration), and identified a $\sim$50~km spatial threshold for mega urban heat dome formation. Under review at \textit{Science Advances}.

        \item \textbf{HyperClim-Former: 5-metre thermal stress mapping and climate-just routing.} Developed a physics-guided deep learning framework coupling mesoscale forcings with 3D urban morphology for 5~m, 10-min UTCI reconstruction (RMSE~=~1.00\textdegree C); integrated with agent-based courier simulations and multi-objective Pareto optimization to reduce heat stress by 21.1\% with only 4.3~min added delay per shift. Submitted to \textit{npj Urban Sustainability}.

        \item \textbf{Physics-informed STGNN for station-to-grid urban weather nowcasting.} Designed five physically motivated graph topologies (proximity, thermal similarity, wind advection, terrain, LCZ) for 333~m station-to-grid downscaling; implemented WRF-to-observation transfer learning with thermodynamic constraints, reducing WRF biases by 25--67\% while retaining 92.7\% spatial heterogeneity. Submitted to \textit{Atmospheric Chemistry and Physics}.
    \end{itemize}

    \item \textbf{Cumulative Effects of Hydrology and Ecology in the Yellow River Basin} \hfill 2022--2023\\
    Project Lead, Northwest A\&F University
    \begin{itemize}
        \item Revealed the hydrology--ecosystem coupling effect in the Yellow River Basin, established a large-scale cumulative effect model, and developed remote sensing data pipelines to support the research findings.
    \end{itemize}

    \item \textbf{Development and Design of Robot Visual Aid System} \hfill 2021--2022\\
    Core Member, Northwest A\&F University
    \begin{itemize}
        \item Programmed and debugged OpenCV-based real-time object detection under Linux for RoboMaster college league robotics and agricultural picking robots.
    \end{itemize}

    \item \textbf{Regional Water and Food Big Data Platform} \hfill 2021--2022\\
    Core Member, Northwest A\&F University
    \begin{itemize}
        \item Analyzed regional water consumption patterns and database architecture to map spatiotemporal agricultural water use efficiency.
    \end{itemize}
\end{enumerate}
